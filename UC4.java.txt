/**
 * OOPS Banner Application
 *
 * UC4: Store banner lines in a String array and
 * print them using a loop for better modularity.
 *
 * @version 4.0
 */
public class OOPSBannerApp {

    public static void main(String[] args) {

        String[] bannerLines = {
            String.join("", " *****   ", " *****   ", " ******   ", " ***** "),
            String.join("", "*     * ", "*     * ", "*        ", "*     *"),
            String.join("", "*     * ", "*     * ", "*        ", "*      "),
            String.join("", "*     * ", "*     * ", " ******  ", " ***** "),
            String.join("", "*     * ", "*     * ", "*        ", "      *"),
            String.join("", "*     * ", "*     * ", "*        ", "*     *"),
            String.join("", " *****   ", " *****   ", " *        ", " ***** ")
        };

        for (String line : bannerLines) {
            System.out.println(line);
        }
    }
}
